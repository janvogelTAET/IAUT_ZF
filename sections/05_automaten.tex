\vlsec{cB5}{B5}{Endliche Automaten}
\label{sec:automaten}

\SW{Zustand} Situation, in der sich das System befindet (Init,
Automatik, Hand, Störung). Immer \emph{genau ein} aktiver Zustand.
\SW{Transition} Bedingung für den Wechsel: Sensor, Tastendruck (Flanke!),
abgelaufener Timer, Kombination davon.
\SW{Diagramm} Zustände als Kreis, Transitionen als Pfeil mit Bedingung,
definierter Startzustand (Init). Steuerung, keine Regelung.

\subsection{Umsetzung in ST: ENUM + CASE}
\SW{ENUM} eigener Datentyp (unter DUTs): Namen statt Zahlen, z.\,B.
\R{JANUAR..DEZEMBER} $\to$ 1..12. Nur für die Lesbarkeit, im
Maschinencode stehen Zahlen.
\SW{CASE} springt abhängig vom Wert einer Variable in einen Codeblock
(wie \R{switch}); kompakter und lesbarer als eine \R{IF/ELSIF}-Kette.
Mehrere Werte pro Zweig mit Komma, Rest mit \R{ELSE}.
\begin{lstlisting}
TYPE E_Monat : (JANUAR := 1, FEBRUAR, MAERZ, APRIL, MAI, JUNI, JULI,
                AUGUST, SEPTEMBER, OKTOBER, NOVEMBER, DEZEMBER);
END_TYPE
TYPE E_Jahreszeit : (FRUEHLING := 1, SOMMER, HERBST, WINTER); END_TYPE

CASE eMonat OF
  MAERZ, APRIL, MAI:           eZeit := FRUEHLING;
  JUNI, JULI, AUGUST:          eZeit := SOMMER;
  SEPTEMBER, OKTOBER, NOVEMBER: eZeit := HERBST;
ELSE                           eZeit := WINTER;
END_CASE
\end{lstlisting}
\SW{Automat} Zustandsvariable vom ENUM-Typ hält den aktuellen Zustand,
\R{CASE} springt in dessen Codeblock, Zustandswechsel $=$ einfache
Zuweisung \R{eState := NEU;} \ra wirkt ab dem nächsten Zyklus.

\begin{rezept}[Automat in ST -- Grundschema]
\label{rez:automat}
\begin{enumerate}
\item ENUM mit allen Zuständen unter DUTs (Startzustand als Initialwert).
\item Eingänge einlesen und vorverarbeiten (Flanken, Timer \emph{vor} dem
  \R{CASE} aufrufen).
\item \R{CASE eState OF}: pro Zustand Aktionen, am Ende Transition prüfen
  und \R{eState := NEU;} setzen.
\item Ausgänge \emph{nach} dem \R{CASE} aus dem Zustand ableiten
  (\R{bMotor := eState = E\_RUN;}) \ra kein Ausgang bleibt hängen.
\item Logik in einen \R{FUNCTION\_BLOCK}, in \R{MAIN} instanziieren
  \ra mehrfach verwendbar.
\end{enumerate}
\end{rezept}
\begin{falle}[Typische Fehler]
Taster ohne Flanke abfragen \ra Zustand rast in einem Zyklus durch
mehrere Zustände. Timer nur im \R{CASE}-Zweig aufrufen \ra läuft nicht
weiter. Störung nur in einem Zustand prüfen \ra wird übersehen (globale
Transition \emph{vor} dem \R{CASE}).
\end{falle}

\label{bsp:foerderband}

\begin{lstlisting}
TYPE EN_FOERDERBAND_ZUSTAND :
  (INIT, READY, RUNNING, STOPPED, FAULT);
END_TYPE

FUNCTION_BLOCK FB_FOERDERBAND_LOGIK
VAR_INPUT
  StartButton, StopButton, FaultSignal, ResetButton : BOOL;
END_VAR
VAR_OUTPUT ConveyorOn, FaultIndicator : BOOL; END_VAR
VAR
  eZustand : EN_FOERDERBAND_ZUSTAND := INIT;
  rtStart, rtReset : R_TRIG;
END_VAR

rtStart(CLK := StartButton);
rtReset(CLK := ResetButton);
IF FaultSignal THEN eZustand := FAULT; END_IF  // gilt überall

CASE eZustand OF
  INIT:    eZustand := READY;
  READY:   IF rtStart.Q THEN eZustand := RUNNING; END_IF
  RUNNING: IF StopButton THEN eZustand := STOPPED; END_IF
  STOPPED: IF NOT StopButton THEN eZustand := READY; END_IF
  FAULT:   IF rtReset.Q AND NOT FaultSignal THEN
             eZustand := READY;
           END_IF
END_CASE

ConveyorOn     := (eZustand = RUNNING);
FaultIndicator := (eZustand = FAULT);
\end{lstlisting}
\begin{lstlisting}
PROGRAM MAIN
VAR
  bStart  AT %I* : BOOL;   bStop  AT %I* : BOOL;
  bFault  AT %I* : BOOL;   bReset AT %I* : BOOL;
  bMotor  AT %Q* : BOOL;   bLampe AT %Q* : BOOL;
  fbBand1 : FB_FOERDERBAND_LOGIK;
END_VAR
fbBand1(StartButton := bStart, StopButton := bStop,
        FaultSignal := bFault, ResetButton := bReset,
        ConveyorOn => bMotor, FaultIndicator => bLampe);
\end{lstlisting}
\begin{formel}[(e) Zweites Förderband]
\label{falle:instanz}
Zweite \T{Instanz} \R{fbBand2 : FB\_FOERDERBAND\_LOGIK;} plus eigene
E/A-Variablen (\R{AT \%I*}/\R{AT \%Q*}) deklarieren, verknüpfen und
\R{fbBand2(...)} aufrufen. Der FB-Code bleibt unverändert -- jede Instanz
hat ihren eigenen \R{eZustand}. Genau darum gehört die Logik in einen FB
und nicht ins Programm.
\end{formel}

\subsection{MEP HS24: Ampel für eine Kreuzung}
\label{bsp:ampel}
\begin{lstlisting}
TYPE EN_FARBEN        : (ROT, GELB, GRUEN);      END_TYPE
TYPE EN_BETRIEBSMODUS : (AUTOMATIK, MANUELL);    END_TYPE
TYPE EN_PHASE : (A_GRUEN, A_GELB, B_GRUEN, B_GELB); END_TYPE

VAR_GLOBAL   // GVL_IO
  bA_Rot AT %Q* : BOOL; bA_Gelb AT %Q* : BOOL; bA_Gruen AT %Q* : BOOL;
  bB_Rot AT %Q* : BOOL; bB_Gelb AT %Q* : BOOL; bB_Gruen AT %Q* : BOOL;
  bTA_Rot AT %I* : BOOL; bTA_Gelb AT %I* : BOOL; bTA_Gruen AT %I* : BOOL;
  bTB_Rot AT %I* : BOOL; bTB_Gelb AT %I* : BOOL; bTB_Gruen AT %I* : BOOL;
  bHand   AT %I* : BOOL;              // Wahlschalter Handbetrieb
END_VAR

PROGRAM MAIN
VAR
  eModus  : EN_BETRIEBSMODUS := AUTOMATIK;
  arAmpel : ARRAY[1..2] OF EN_FARBEN := [ROT, ROT];
  ePhase  : EN_PHASE := A_GRUEN;
  tonGruen, tonGelb : TON;
END_VAR

IF GVL_IO.bHand THEN eModus := MANUELL; ELSE eModus := AUTOMATIK; END_IF

CASE eModus OF
AUTOMATIK:
  tonGruen(IN := (ePhase = A_GRUEN) OR (ePhase = B_GRUEN), PT := T#10S);
  tonGelb (IN := (ePhase = A_GELB)  OR (ePhase = B_GELB),  PT := T#3S);
  CASE ePhase OF
    A_GRUEN: arAmpel[1] := GRUEN; arAmpel[2] := ROT;
             IF tonGruen.Q THEN ePhase := A_GELB; END_IF
    A_GELB:  arAmpel[1] := GELB;  arAmpel[2] := ROT;
             IF tonGelb.Q THEN ePhase := B_GRUEN; END_IF
    B_GRUEN: arAmpel[1] := ROT;   arAmpel[2] := GRUEN;
             IF tonGruen.Q THEN ePhase := B_GELB; END_IF
    B_GELB:  arAmpel[1] := ROT;   arAmpel[2] := GELB;
             IF tonGelb.Q THEN ePhase := A_GRUEN; END_IF
  END_CASE
MANUELL:   // Grün nur, wenn die andere Ampel nicht grün ist
  IF GVL_IO.bTA_Rot THEN arAmpel[1] := ROT;
  ELSIF GVL_IO.bTA_Gelb THEN arAmpel[1] := GELB;
  ELSIF GVL_IO.bTA_Gruen AND arAmpel[2] <> GRUEN THEN arAmpel[1] := GRUEN;
  END_IF
  IF GVL_IO.bTB_Rot THEN arAmpel[2] := ROT;
  ELSIF GVL_IO.bTB_Gelb THEN arAmpel[2] := GELB;
  ELSIF GVL_IO.bTB_Gruen AND arAmpel[1] <> GRUEN THEN arAmpel[2] := GRUEN;
  END_IF
END_CASE

GVL_IO.bA_Rot := arAmpel[1] = ROT;  GVL_IO.bA_Gelb := arAmpel[1] = GELB;
GVL_IO.bA_Gruen := arAmpel[1] = GRUEN;
GVL_IO.bB_Rot := arAmpel[2] = ROT;  GVL_IO.bB_Gelb := arAmpel[2] = GELB;
GVL_IO.bB_Gruen := arAmpel[2] = GRUEN;
\end{lstlisting}
\begin{falle}[Ampel: was man begründen sollte]
Zwei Timer reichen, weil zwischen zwei Grünphasen immer eine Gelbphase
liegt (\R{IN} fällt dazwischen ab \ra Reset). Rot einer Ampel dauert hier
Grün $+$ Gelb der anderen $=13$\,s; exakt 10\,s Rot ginge nur mit
kürzerer Grünphase der Gegenrichtung. „Nie beide grün“ ist in Automatik
durch die Phasen garantiert, im Handbetrieb durch die Verriegelung.
\end{falle}

\subsection{AS 4: Safe (Tresor) in ST}
\label{bsp:safe}
Init (Code-Array füllen) \ra Close (Zähler/Lampen reset) \ra CheckDigit
(nur bei \R{risingEdge}, 4 Tasten) \ra Open (grün, \R{lightDuration}
5\,s) bzw.\ WrongCode (rot, 5\,s) \ra Close. \R{TRUE} $=$ blau,
\R{FALSE} $=$ gelb.
\begin{lstlisting}
TYPE E_SafeState : (S_INIT, S_CLOSE, S_CHECK, S_OPEN, S_WRONG); END_TYPE

PROGRAM P_Safe
VAR
  eState       : E_SafeState := S_INIT;
  fbBlau, fbGelb : EdgeDetector;          // aus AS 2 (PLCopenXML-Import)
  codeArray    : ARRAY[0..3] OF BOOL;
  currentIndex, correctCounter : INT;
  tonLicht     : TON;
  bTaste, bBlau : BOOL;
END_VAR
VAR CONSTANT lightDuration : TIME := T#5S; END_VAR

// 1) Eingaenge + Timer VOR dem CASE
fbBlau(inputValue := GVL.bTasterBlau);
fbGelb(inputValue := GVL.bTasterGelb);
bTaste := fbBlau.risingEdge OR fbGelb.risingEdge;
bBlau  := fbBlau.risingEdge;
tonLicht(IN := (eState = S_OPEN) OR (eState = S_WRONG), PT := lightDuration);

// 2) Zustaende
CASE eState OF
S_INIT:
  codeArray[0] := TRUE;  codeArray[1] := FALSE;   // blau, gelb,
  codeArray[2] := FALSE; codeArray[3] := TRUE;    // gelb, blau
  eState := S_CLOSE;
S_CLOSE:
  currentIndex := 0; correctCounter := 0;
  eState := S_CHECK;
S_CHECK:
  IF bTaste THEN
    IF bBlau = codeArray[currentIndex] THEN
      correctCounter := correctCounter + 1;
    END_IF
    currentIndex := currentIndex + 1;   // ERST nach dem Vergleich
    IF currentIndex >= 4 THEN
      IF correctCounter = 4 THEN eState := S_OPEN;
      ELSE                       eState := S_WRONG; END_IF
    END_IF
  END_IF
S_OPEN, S_WRONG:
  IF tonLicht.Q THEN eState := S_CLOSE; END_IF
END_CASE

// 3) Ausgaenge NACH dem CASE aus dem Zustand
GVL.bLampeGruen := (eState = S_OPEN);
GVL.bLampeRot   := (eState = S_WRONG);
\end{lstlisting}
\begin{falle}[Safe]
Index zu früh erhöhen \ra falsches Array-Element verglichen bzw.\
Zugriff auf \R{codeArray[4]} (ausserhalb). Ohne Flanke zählt ein
gedrückter Taster in jedem Zyklus. Nach falschem Code nicht sofort
abbrechen, sondern erst nach 4 Tasten \ra verrät nicht, welche Stelle
falsch war.
\end{falle}
\SW{Zusatz} \R{doubleTap} blau bricht die Eingabe ab (\ra S\_CLOSE). Im
Zustand Open beide Tasten gleichzeitig \ra neuer Zustand
\R{S\_PROGRAM}, der 4 Tasten in \R{codeArray} schreibt.

\subsection{AS 4: Ampel als Funktionsblock}
\label{bsp:ampelfb}
Init \ra Red \ra(clear) Red\&Yellow \ra(2\,s) Green \ra(block) Yellow
\ra(5\,s) Red. Aus Red mit \R{disable\_system} \ra Disabled (gelb blinkt
im 1-s-Takt) \ra zurück zu Red, wenn \R{disable} weg.
\begin{lstlisting}
TYPE E_Ampel : (A_INIT, A_RED, A_REDYELLOW, A_GREEN, A_YELLOW, A_DISABLED);
END_TYPE

FUNCTION_BLOCK FB_Ampel
VAR_INPUT  clear_road, block_road, disable_system : BOOL; END_VAR
VAR_OUTPUT red, yellow, green : BOOL; END_VAR
VAR
  eState : E_Ampel := A_INIT;
  tonRY, tonY, tonBlink : TON;
  bBlink : BOOL;
END_VAR

tonRY(IN := eState = A_REDYELLOW, PT := T#2S);
tonY (IN := eState = A_YELLOW,    PT := T#5S);
// Taktgeber: TON setzt sich mit NOT Q selbst zurueck
tonBlink(IN := (eState = A_DISABLED) AND NOT tonBlink.Q, PT := T#1S);
IF tonBlink.Q THEN bBlink := NOT bBlink; END_IF

CASE eState OF
A_INIT:      eState := A_RED;
A_RED:       IF disable_system THEN eState := A_DISABLED;
             ELSIF clear_road  THEN eState := A_REDYELLOW; END_IF
A_REDYELLOW: IF tonRY.Q THEN eState := A_GREEN; END_IF
A_GREEN:     IF block_road THEN eState := A_YELLOW; END_IF
A_YELLOW:    IF tonY.Q THEN eState := A_RED; END_IF
A_DISABLED:  IF NOT disable_system THEN eState := A_RED; END_IF
END_CASE

red    := (eState = A_RED) OR (eState = A_REDYELLOW);
yellow := (eState = A_REDYELLOW) OR (eState = A_YELLOW)
          OR ((eState = A_DISABLED) AND bBlink);
green  := (eState = A_GREEN);
\end{lstlisting}
\begin{lstlisting}
PROGRAM MAIN
VAR
  clear, block, disable : BOOL;      // Kontrollvariablen (Visu-Taster)
  red, yellow, green    : BOOL;      // Ausgaenge (Visu-Lampen)
  ampel_1 : FB_Ampel;
END_VAR
ampel_1(clear_road := clear, block_road := block, disable_system := disable,
        red => red, yellow => yellow, green => green);
\end{lstlisting}
\SW{Visualisierung} VISUs \ra neue Visu; aus der Toolbox Lampen
(\R{Lamp1}) und Taster (\R{Button}), in den Eigenschaften mit
\R{MAIN.clear} bzw.\ \R{MAIN.red} verknüpfen.
\SW{Kreuzung} Weil die Ampel ein FB ist: 4 Instanzen. Phase A (Ampel 1+3
grün, 5\,s) \ra sicherer Wechsel (alle erst über Gelb auf Rot) \ra Phase B
(2+4 grün, 8\,s) \ra zurück. Zusatz: Taster in der Visu deaktiviert die
Anlage (\R{disable\_system} an alle Instanzen).
